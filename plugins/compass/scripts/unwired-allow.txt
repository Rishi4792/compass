# a gate that never runs. From the outside the two look identical, which is how six gates in this
# plugin came to exist and never fire. This file is where the difference is DECLARED, so the
# distinction is a recorded decision instead of a shrug.
#

worktree-rm         # CLI: destructive, deliberately never automatic
abort-clear         # CLI: a person clearing an abort sentinel they set
auto-init           # orchestrator setup, invoked by auto-start, not standalone
budget-init         # orchestrator setup, invoked by auto-start, not standalone
auto-resume         # entered by the resume command's prose, not by a script
